/**
 * @jest-environment node
 *
 * BACKLOG-4243 — GENERATOR for profile-v79.sql. Not a test of this tree; it runs against the
 * v2.48.0 TAG's own code. Kept as .txt so no jest glob, tsc or eslint run in this tree picks it up.
 *
 * Regenerate (needs a Node-ABI build of better-sqlite3-multiple-ciphers in node_modules):
 *   git worktree add --detach ../Mad-v248 v2.48.0
 *   ln -s <a node_modules with a Node-ABI driver> ../Mad-v248/node_modules
 *   cp electron/services/__tests__/fixtures/v2.48.0/generateProfileV79.gen.ts.txt \
 *      ../Mad-v248/electron/services/__tests__/generateProfileV79.gen.test.ts
 *   (cd ../Mad-v248 && PROFILE_V79_OUT=$PWD/../profile-v79.sql \
 *      npx jest electron/services/__tests__/generateProfileV79.gen.test.ts)
 *   cp ../profile-v79.sql electron/services/__tests__/fixtures/v2.48.0/profile-v79.sql
 *
 * What it does: the tag's databaseService.initialize() on an EMPTY profile (schema.sql seeds 70, the
 * tag's chain 71..79 runs), then dumps the resulting file: every sqlite_master object (tables,
 * indexes, views, triggers, each group in sqlite_master order) and every row of every table.
 */

it("dumps the profile the v2.48.0 build creates", async () => {
  userDataDir = fs.mkdtempSync(path.join(os.tmpdir(), "keepr-gen248-"));
  // eslint-disable-next-line @typescript-eslint/no-require-imports
  require("../../../tests/helpers/installTestCapabilities").installTestCapabilities();
  // eslint-disable-next-line @typescript-eslint/no-require-imports
  const svc = require("../databaseService").default;
  await expect(svc.initialize()).resolves.toBe(true);
  const db = svc.db as DatabaseType;
  const head = execSync("git rev-parse HEAD", { encoding: "utf8" }).trim();
  const blob = execSync("git rev-parse HEAD:electron/database/schema.sql", { encoding: "utf8" }).trim();
  const version = (db.prepare("SELECT version FROM schema_version WHERE id = 1").get() as { version: number }).version;
  const userVersion = db.pragma("user_version", { simple: true });
  const objs = db
    .prepare("SELECT type, name, tbl_name, sql FROM sqlite_master WHERE sql IS NOT NULL AND name NOT LIKE 'sqlite_%' ORDER BY rowid")
    .all() as Array<{ type: string; name: string; tbl_name: string; sql: string }>;
  const out: string[] = [
    "-- BACKLOG-4243 fixture: the database a v2.48.0 build creates on an empty profile.",
    `-- Generated by electron/services/__tests__/fixtures/v2.48.0/generateProfileV79.gen.ts.txt at tag commit ${head}`,
    `-- (v2.48.0 electron/database/schema.sql blob ${blob}); regeneration command in that file's header.`,
    `-- schema_version ${version}; PRAGMA user_version ${userVersion}; ${objs.length} objects. Do not edit by hand.`,
  ];
  for (const type of ["table", "index", "view", "trigger"]) {
    for (const o of objs.filter((x) => x.type === type)) out.push(`${o.sql};`);
  }
  for (const t of objs.filter((x) => x.type === "table")) {
    const cols = (db.prepare(`PRAGMA table_info("${t.name}")`).all() as Array<{ name: string }>).map((c) => `quote("${c.name}")`);
    const rows = db.prepare(`SELECT ${cols.join(" || ',' || ")} AS v FROM "${t.name}" ORDER BY rowid`).all() as Array<{ v: string }>;
    for (const r of rows) out.push(`INSERT INTO "${t.name}" VALUES (${r.v});`);
  }
  const seq = db.prepare("SELECT name FROM sqlite_master WHERE name = 'sqlite_sequence'").get();
  if (seq) {
    for (const r of db.prepare("SELECT name, seq FROM sqlite_sequence").all() as Array<{ name: string; seq: number }>) {
      out.push(`INSERT INTO sqlite_sequence (name, seq) VALUES ('${r.name}', ${r.seq});`);
    }
  }
  out.push(`PRAGMA user_version = ${userVersion};`);
  fs.writeFileSync(process.env.PROFILE_V79_OUT as string, out.join("\n") + "\n");
  // Round trip: a database built from the dump has the same objects, the same DDL and the same rows.
  // eslint-disable-next-line @typescript-eslint/no-require-imports
  const Real = require("../../../node_modules/better-sqlite3-multiple-ciphers");
  const copy = new Real(":memory:") as DatabaseType;
  copy.exec(fs.readFileSync(process.env.PROFILE_V79_OUT as string, "utf8"));
  const master = (d: DatabaseType) =>
    d.prepare("SELECT type, name, tbl_name, sql FROM sqlite_master WHERE name NOT LIKE 'sqlite_autoindex_%' ORDER BY type, name").all();
  expect(master(copy)).toEqual(master(db));
  for (const t of objs.filter((x) => x.type === "table")) {
    expect(copy.prepare(`SELECT * FROM "${t.name}" ORDER BY rowid`).all()).toEqual(db.prepare(`SELECT * FROM "${t.name}" ORDER BY rowid`).all());
  }
  // For the record: objects the tag's chain (71..79) adds beyond its schema.sql alone.
  const bare = new Real(":memory:") as DatabaseType;
  bare.exec(fs.readFileSync(path.join(__dirname, "..", "..", "database", "schema.sql"), "utf8"));
  const key = (r: unknown) => JSON.stringify(r);
  const bareSet = new Set((master(bare) as unknown[]).map(key));
  const chainOnly = (master(db) as Array<{ type: string; name: string }>).filter((r) => !bareSet.has(key(r))).map((r) => `${r.type}:${r.name}`);
  process.stderr.write(`CHAIN-ONLY OBJECTS: ${JSON.stringify(chainOnly)}\n`);
  copy.close();
  bare.close();
  await svc.close();
  fs.rmSync(userDataDir, { recursive: true, force: true });
});
